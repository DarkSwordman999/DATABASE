 RETURNS rational
 LANGUAGE sql

  SELECT CASE WHEN x.a = 0 THEN rat(0) ELSE rat(x.b, x.a) END
$function$
CREATE OR REPLACE FUNCTION public.rat_mul(x rational, y rational)
 RETURNS rational
 LANGUAGE plpgsql
 IMMUTABLE STRICT
AS $function$
DECLARE g1 float := greatest(rat_gcd(x.a, y.b), 1);
        g2 float := greatest(rat_gcd(y.a, x.b), 1);
BEGIN
  RETURN rat((x.a / g1) * (y.a / g2), (x.b / g2) * (y.b / g1));
END $function$
