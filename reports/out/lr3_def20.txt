CREATE OR REPLACE FUNCTION public.v3_cross(a vector3, b vector3)
 RETURNS vector3
 LANGUAGE sql
 IMMUTABLE STRICT
AS $function$
  SELECT ROW(a.y * b.z - a.z * b.y,
             a.z * b.x - a.x * b.z,
             a.x * b.y - a.y * b.x)::vector3
$function$
